func calcular_fibonacci(n: int) -> int:
    if n <= 1:
        return n
    return calcular_fibonacci(n - 1) + calcular_fibonacci(n - 2)

# Ponto de entrada principal
pub func main() -> void:
    res: int = calcular_fibonacci(10)
    print(f"Resultado otimizado: {res}")
